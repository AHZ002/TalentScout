"""Manage interview-session lifecycle and validated durable InterviewState persistence."""

from uuid import UUID

from talentscout.agents.state import InterviewState
from talentscout.agents.state_serializer import (
    deserialize_interview_state,
    serialize_interview_state,
)
from talentscout.db.models.interview_session import (
    InterviewSession,
    InterviewSessionStatus,
)
from talentscout.jobs.repositories.interview_session import InterviewSessionRepository
from talentscout.jobs.repositories.job import JobRepository


class JobNotFoundError(LookupError):
    """Raised when an interview is requested for a nonexistent job."""


class InterviewSessionNotFoundError(LookupError):
    """Raised when an interview session does not exist."""


class InterviewSessionAlreadyCompletedError(ValueError):
    """Raised when a caller attempts to resume or modify a completed interview."""


class InvalidInterviewStateError(ValueError):
    """Raised when interview state conflicts with its session or lifecycle."""


class InterviewSessionService:
    """Coordinate job validation, state validation, and session persistence."""

    def __init__(
        self,
        session_repository: InterviewSessionRepository,
        job_repository: JobRepository,
    ) -> None:
        """Initialize the service with repositories sharing the request session."""
        self.session_repository = session_repository
        self.job_repository = job_repository

    async def create_session(self, job_id: UUID) -> InterviewSession:
        """Create an in-progress interview initialized from the selected job."""
        job = await self.job_repository.get_by_id(job_id)
        if job is None:
            raise JobNotFoundError(f"Job '{job_id}' was not found.")

        initial_state: InterviewState = {
            "job_id": job.id,
            "job_description": job.description,
            "interview_history": [],
            "retrieved_context": [],
        }
        validated_state = self._validate_state(
            initial_state,
            expected_job_id=job.id,
        )

        interview_session = InterviewSession(
            job_id=job.id,
            status=InterviewSessionStatus.IN_PROGRESS,
            state=serialize_interview_state(validated_state),
        )
        return await self.session_repository.create(interview_session)

    async def get_session(self, session_id: UUID) -> InterviewSession | None:
        """Return a session after validating its persisted state and lifecycle."""
        interview_session = await self.session_repository.get_by_id(session_id)
        if interview_session is None:
            return None

        state = self._read_session_state(interview_session)
        self._validate_lifecycle(interview_session.status, state)
        return interview_session

    async def resume_session(
        self,
        session_id: UUID,
    ) -> tuple[InterviewSession, InterviewState]:
        """Load validated state for an in-progress interview graph invocation."""
        interview_session = await self.get_session(session_id)
        if interview_session is None:
            raise InterviewSessionNotFoundError(
                f"Interview session '{session_id}' was not found."
            )

        if interview_session.status == InterviewSessionStatus.COMPLETED:
            raise InterviewSessionAlreadyCompletedError(
                f"Interview session '{session_id}' is already completed."
            )

        return interview_session, self._read_session_state(interview_session)

    async def save_state(
        self,
        session_id: UUID,
        state: InterviewState,
    ) -> InterviewSession:
        """Validate and persist graph state, updating status when a report exists."""
        interview_session = await self.get_session(session_id)
        if interview_session is None:
            raise InterviewSessionNotFoundError(
                f"Interview session '{session_id}' was not found."
            )

        if interview_session.status == InterviewSessionStatus.COMPLETED:
            raise InterviewSessionAlreadyCompletedError(
                f"Interview session '{session_id}' is already completed."
            )

        previous_state = self._read_session_state(interview_session)
        validated_state = self._validate_state(
            state,
            expected_job_id=interview_session.job_id,
        )

        # The job description is the source-of-truth snapshot captured at session creation.
        if validated_state["job_description"] != previous_state["job_description"]:
            raise InvalidInterviewStateError(
                "The job description cannot change during an interview session."
            )

        self._validate_lifecycle(
            InterviewSessionStatus.COMPLETED
            if validated_state.get("interview_complete", False)
            else InterviewSessionStatus.IN_PROGRESS,
            validated_state,
        )

        interview_session.state = serialize_interview_state(validated_state)
        interview_session.status = (
            InterviewSessionStatus.COMPLETED
            if validated_state.get("interview_complete", False)
            else InterviewSessionStatus.IN_PROGRESS
        )
        return await self.session_repository.update(interview_session)

    @staticmethod
    def _validate_state(
        state: InterviewState,
        expected_job_id: UUID,
    ) -> InterviewState:
        """Round-trip state through Pydantic and enforce required session identity."""
        validated_state = deserialize_interview_state(
            serialize_interview_state(state)
        )
        state_job_id = validated_state.get("job_id")
        job_description = validated_state.get("job_description")

        if state_job_id is None or job_description is None or not job_description.strip():
            raise InvalidInterviewStateError(
                "Interview state must include a job ID and a non-empty job description."
            )
        if state_job_id != expected_job_id:
            raise InvalidInterviewStateError(
                "Interview state job ID does not match the session's job ID."
            )

        return validated_state

    @classmethod
    def _read_session_state(
        cls,
        interview_session: InterviewSession,
    ) -> InterviewState:
        """Deserialize stored JSON and verify it belongs to this database session."""
        try:
            state = deserialize_interview_state(interview_session.state)
        except (TypeError, ValueError) as exc:
            raise InvalidInterviewStateError(
                "Persisted interview state is invalid."
            ) from exc

        return cls._validate_state(state, expected_job_id=interview_session.job_id)

    @staticmethod
    def _validate_lifecycle(
        status_value: InterviewSessionStatus | str,
        state: InterviewState,
    ) -> None:
        """Ensure completion status agrees with the final state and report."""
        interview_complete = state.get("interview_complete", False)
        has_report = "assessment_report" in state

        if interview_complete != has_report:
            raise InvalidInterviewStateError(
                "Interview completion and assessment-report state are inconsistent."
            )

        state_is_completed = interview_complete and has_report
        session_is_completed = status_value == InterviewSessionStatus.COMPLETED
        if state_is_completed != session_is_completed:
            raise InvalidInterviewStateError(
                "Interview session status does not match its persisted state."
            )

========== backend/src/talentscout/jobs/repositories/interview_session.py ==========

"""Repository for durable candidate interview sessions."""

from uuid import UUID

from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from talentscout.db.models.interview_session import InterviewSession


class InterviewSessionRepository:
    """Handle database persistence for interview sessions."""

    def __init__(self, session: AsyncSession) -> None:
        self.session = session

    async def create(
        self,
        interview_session: InterviewSession,
    ) -> InterviewSession:
        """Persist a new interview session."""
        self.session.add(interview_session)
        await self.session.flush()
        await self.session.refresh(interview_session)
        return interview_session

    async def get_by_id(
        self,
        session_id: UUID,
    ) -> InterviewSession | None:
        """Find an interview session by its identifier."""
        result = await self.session.execute(
            select(InterviewSession).where(
                InterviewSession.id == session_id
            )
        )
        return result.scalar_one_or_none()

    async def update(
        self,
        interview_session: InterviewSession,
    ) -> InterviewSession:
        """Persist changes to an existing interview session."""
        await self.session.flush()
        await self.session.refresh(interview_session)
        return interview_session

from pydantic import BaseModel, ConfigDict, Field, field_validator


class InterviewAnswerSubmit(BaseModel):
    """Validate a candidate's answer submitted to an interview session."""

    model_config = ConfigDict(extra="forbid")

    candidate_answer: str = Field(min_length=1)

    @field_validator("candidate_answer")
    @classmethod
    def normalize_candidate_answer(cls, value: str) -> str:
        """Trim surrounding whitespace and reject blank answers."""
        normalized = value.strip()

        if not normalized:
            raise ValueError("Candidate answer must not be blank.")

        return normalized

"""SQLAlchemy model for a durable candidate interview session."""

from datetime import UTC, datetime
from enum import StrEnum
from uuid import UUID, uuid4

from sqlalchemy import DateTime, ForeignKey, String
from sqlalchemy.dialects.postgresql import JSONB
from sqlalchemy.orm import Mapped, mapped_column

from talentscout.db.base import Base


class InterviewSessionStatus(StrEnum):
    """Lifecycle states for a candidate interview session."""

    IN_PROGRESS = "in_progress"
    COMPLETED = "completed"


class InterviewSession(Base):
    """Persist the complete LangGraph state for one candidate interview."""

    __tablename__ = "interview_sessions"

    id: Mapped[UUID] = mapped_column(
        primary_key=True,
        default=uuid4,
    )
    job_id: Mapped[UUID] = mapped_column(
        ForeignKey("jobs.id", ondelete="CASCADE"),
        nullable=False,
        index=True,
    )
    status: Mapped[InterviewSessionStatus] = mapped_column(
        String(20),
        default=InterviewSessionStatus.IN_PROGRESS,
        nullable=False,
    )
    state: Mapped[dict[str, object]] = mapped_column(
        JSONB,
        nullable=False,
    )
    created_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True),
        default=lambda: datetime.now(UTC),
        nullable=False,
    )
    updated_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True),
        default=lambda: datetime.now(UTC),
        onupdate=lambda: datetime.now(UTC),
        nullable=False,
    )

"""State shared between nodes in the TalentScout interview graph."""

from __future__ import annotations

from typing import TypedDict
from uuid import UUID

from talentscout.agents.answer_evaluation import AnswerEvaluation
from talentscout.agents.assessment_report import AssessmentReport
from talentscout.agents.interview_types import InterviewTurn
from talentscout.agents.role_competency import RoleCompetencyAnalysis


class InterviewState(TypedDict, total=False):
    """Data passed between interview workflow agents."""

    # Identifies the hiring job whose context is being used.
    job_id: UUID

    # The job description defines the skills and requirements being assessed.
    job_description: str

    # Candidate's latest answer, when available.
    candidate_answer: str

    # Structured evaluation of the candidate's latest answer.
    answer_evaluation: AnswerEvaluation

    # Competencies addressed across completed interview turns.
    assessed_competencies: list[str]

    # Competencies that still need interview evidence.
    remaining_competencies: list[str]    

    # Indicates that the caller has submitted the final interview answer.
    interview_complete: bool

    # Final assessment produced after the interview is completed.
    assessment_report: AssessmentReport 

    # Document text retrieved using semantic search.
    retrieved_context: list[str]

    # Question generated by the interviewer agent.
    current_question: str

    # Completed question-and-answer exchanges from the interview.
    interview_history: list[InterviewTurn]    

    # Optional Additional Interview Guidance supplied to role analysis.
    additional_guidance: list[str]

    # Structured competencies extracted from the JD and optional guidance.
    role_competency_analysis: RoleCompetencyAnalysis

"""Serialize and validate InterviewState for durable persistence."""

from typing import cast

from pydantic import TypeAdapter

from talentscout.agents.state import InterviewState

_INTERVIEW_STATE_ADAPTER = TypeAdapter(InterviewState)


def serialize_interview_state(state: InterviewState) -> dict[str, object]:
    """Convert an InterviewState into JSON-compatible data."""
    serialized = _INTERVIEW_STATE_ADAPTER.dump_python(
        state,
        mode="json",
    )
    return cast(dict[str, object], serialized)


def deserialize_interview_state(payload: object) -> InterviewState:
    """Validate and reconstruct InterviewState from persisted data."""
    return _INTERVIEW_STATE_ADAPTER.validate_python(payload)
